```javascript
// В PetMiniGameModal.js
const createOptimizedGameLoop = (updateCallback, renderCallback) => {
  // Простая реализация без оптимизаций
  let animationId = null;
  let lastUpdateTime = 0;
  
  const gameLoop = (currentTime) => {
    if (!lastUpdateTime) lastUpdateTime = currentTime;
    const deltaTime = currentTime - lastUpdateTime;
    updateCallback(deltaTime);
    renderCallback();
    lastUpdateTime = currentTime;
    animationId = requestAnimationFrame(gameLoop);
  };
  
  return { 
    start: () => { if (!animationId) { lastUpdateTime = 0; animationId = requestAnimationFrame(gameLoop); } },
    stop: () => { if (animationId) { cancelAnimationFrame(animationId); animationId = null; } }
  };
};
```
